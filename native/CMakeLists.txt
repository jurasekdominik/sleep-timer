cmake_minimum_required(VERSION 3.20)
project(SleepTimerTaskbar LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The native taskbar host requires the Windows SDK and MSVC")
endif()
add_executable(SleepTimer.Taskbar WIN32 taskbar.cpp)
target_compile_features(SleepTimer.Taskbar PRIVATE cxx_std_17)
target_compile_definitions(SleepTimer.Taskbar PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(SleepTimer.Taskbar PRIVATE /W4 /WX /permissive- /sdl /guard:cf /utf-8)
target_link_options(SleepTimer.Taskbar PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(SleepTimer.Taskbar PRIVATE user32 gdi32 d2d1 dwrite shell32)
set_property(TARGET SleepTimer.Taskbar PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(SleepTimer.Taskbar PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
target_sources(SleepTimer.Taskbar PRIVATE host.manifest)
